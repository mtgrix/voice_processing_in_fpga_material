% Figure 1.3a: Giao thoa kế vuông pha sóng dò (Đo thực nghiệm từ datasets/sample_tone440.wav)
\begin{tikzpicture}[
  >=Stealth,
  font=\small,
  blk/.style={draw=black!75, rounded corners=3pt, fill=blue!8, semithick, align=center, inner sep=3pt},
  mathnode/.style={draw=black!75, circle, fill=yellow!20, thick, inner sep=1pt, minimum size=6.0mm},
  accum/.style={draw=black!75, rounded corners=3pt, fill=teal!12, semithick, align=center, inner sep=3pt},
  outnode/.style={draw=black!85, rounded corners=3pt, fill=green!12, thick, align=center, inner sep=3pt},
  wavebox/.style={draw=black!60, fill=white, thin, inner sep=2pt, align=center},
  arrow/.style={->, thick, black!75}
]

% Khung đầu vào từ file WAV thật
\node[blk, fill=blue!10, text width=2.5cm] (input) at (1.3, 0) {
  \textbf{Khung sóng âm}\\
  $x_\ell[n]$\\
  \scriptsize \nolinkurl{sample_tone440.wav}\\
  $L{=}400,\, f{=}440\text{ Hz}$
};

% Điểm rẽ nhánh
\coordinate (split) at (2.9, 0);
\filldraw[black!80] (split) circle (2.0pt);
\draw[arrow] (input) -- (split);

% Tách hai nhánh trực giao
\coordinate (c_in) at (2.9, 1.1);
\coordinate (s_in) at (2.9, -1.1);
\draw[thick, black!75] (split) -- (c_in);
\draw[thick, black!75] (split) -- (s_in);

% Nhánh trên: Cosine (Đồng pha)
\node[mathnode] (mul_cos) at (4.7, 1.1) {\large $\times$};
\draw[arrow] (c_in) -- (mul_cos);

\node[wavebox, text width=2.6cm] (probe_cos) at (4.7, 2.25) {
  \scriptsize\textbf{Sóng dò $\cos$} ($k{=}14$)\\[0.2mm]
  $\cos\left(\frac{2\pi \cdot 14 \cdot n}{512}\right)$\\[0.5mm]
  \scriptsize $437{,}5\text{ Hz}$ ($14 \times 31{,}25$)
};
\draw[arrow, blue!75!black] (probe_cos) -- (mul_cos);

\node[accum, text width=3.0cm] (acc_cos) at (8.0, 1.1) {
  \textbf{Bộ tích lũy $\sum$}\\[0.5mm]
  \footnotesize Gốc: $\mathbf{+105{,}34}$\\[0.2mm]
  \scriptsize Lệch $1/4$ chu kỳ: $-19{,}27$
};
\draw[arrow] (mul_cos) -- (acc_cos);

\node[outnode, text width=2.4cm] (re_out) at (11.3, 1.1) {
  \textbf{Phần thực Re}\\[0.3mm]
  $\text{Re}(X_\ell[14])$\\[0.5mm]
  \footnotesize $\mathbf{+105{,}34}$
};
\draw[arrow] (acc_cos) -- (re_out);

% Nhánh dưới: Sine (Vuông pha)
\node[mathnode] (mul_sin) at (4.7, -1.1) {\large $\times$};
\draw[arrow] (s_in) -- (mul_sin);

\node[wavebox, text width=2.6cm] (probe_sin) at (4.7, -2.25) {
  \scriptsize\textbf{Sóng dò $-\sin$} ($k{=}14$)\\[0.2mm]
  $-\sin\left(\frac{2\pi \cdot 14 \cdot n}{512}\right)$\\[0.5mm]
  \scriptsize Lệch pha $90^\circ$ trực giao
};
\draw[arrow, red!75!black] (probe_sin) -- (mul_sin);

\node[accum, text width=3.0cm] (acc_sin) at (8.0, -1.1) {
  \textbf{Bộ tích lũy $\sum$}\\[0.5mm]
  \footnotesize Gốc: $\mathbf{+20{,}89}$\\[0.2mm]
  \scriptsize Lệch $1/4$ chu kỳ: $\mathbf{+105{,}74}$
};
\draw[arrow] (mul_sin) -- (acc_sin);

\node[outnode, text width=2.4cm] (im_out) at (11.3, -1.1) {
  \textbf{Phần ảo Im}\\[0.3mm]
  $\text{Im}(X_\ell[14])$\\[0.5mm]
  \footnotesize $\mathbf{+20{,}89}$
};
\draw[arrow] (acc_sin) -- (im_out);

% Ghép thành bin phổ phức & công suất
\node[draw=purple!80!black, rounded corners=4pt, fill=purple!8, thick, align=center, text width=3.6cm] (euler) at (14.9, 0) {
  \textbf{Bin phổ phức \& Công suất}\\[0.8mm]
  $X_\ell[14] = 105{,}34 + j\,20{,}89$\\[1mm]
  \footnotesize \textbf{Công suất $|X_\ell[14]|^2$:}\\[0.3mm]
  $105{,}34^2 + 20{,}89^2 = \mathbf{11.532{,}71}$\\[0.8mm]
  \scriptsize (Lệch $1/4$ kỳ: $(-19{,}27)^2 + 105{,}74^2 = 11.552$)
};

\draw[arrow, purple!80!black] (re_out.east) -- ++(0.3, 0) |- (euler.north west);
\draw[arrow, purple!80!black] (im_out.east) -- ++(0.3, 0) |- (euler.south west);

\end{tikzpicture}
