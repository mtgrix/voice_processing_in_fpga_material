% Figure 1.3b: 257 Linear Bins Collapsing Under 80 Mel Triangular Filters
\begin{tikzpicture}[
  >=Stealth,
  scale=0.92, every node/.style={transform shape},
  font=\small,
  tri/.style={draw=teal!85!black, thick, fill=teal!10},
  hightri/.style={draw=purple!85!black, thick, fill=purple!10},
  bin/.style={draw=black!30, thin},
  ax/.style={->, black!75, thick},
  tk/.style={font=\scriptsize, text=black!75},
  ann/.style={font=\scriptsize, text=black!80}
]

% ==================== (a) Full 0 - 8 kHz Spectrum ====================
\begin{scope}[x=0.0017cm, y=1.2cm]
  \node[font=\bfseries\small, anchor=west] at (-120, 2.2) 
    {(a) Mel Filterbank Spacing Across the Full Audio Band ($0\text{ to } 8\text{ kHz}$)};

  % Frequency Axis
  \draw[ax] (-100, 0) -- (8350, 0) node[right, font=\scriptsize\bfseries] {Frequency (Hz)};
  
  \foreach \f in {0, 1000, 2000, 3000, 4000, 5000, 6000, 7000, 8000} {
    \draw[black!60, semithick] (\f, -0.05) -- (\f, 0.05);
  }
  \node[tk, below] at (0, -0.05) {$0$};
  \node[tk, below] at (1000, -0.05) {$1\text{k}$};
  \node[tk, below] at (2000, -0.05) {$2\text{k}$};
  \node[tk, below] at (3000, -0.05) {$3\text{k}$};
  \node[tk, below] at (4000, -0.05) {$4\text{k}$};
  \node[tk, below] at (5000, -0.05) {$5\text{k}$};
  \node[tk, below] at (6000, -0.05) {$6\text{k}$};
  \node[tk, below] at (7000, -0.05) {$7\text{k}$};
  \node[tk, below] at (8000, -0.05) {$8\text{k}$};

  % Underlying Linear FFT Bins (257 bins total, sample representation)
  \fill[black!5] (0, 0) rectangle (8000, 0.12);
  \foreach \b in {0, 32, 64, 96, 128, 160, 192, 224, 256} {
    \pgfmathsetmacro{\hz}{\b * 31.25}
    \draw[bin] (\hz, 0) -- (\hz, 1.4);
    \node[font=\tiny, text=gray!70, anchor=south] at (\hz, 1.4) {$k{=}\b$};
  }

  % Representative Triangular Filters showing non-linear expansion
  % Low freq filters (dense, narrow)
  \draw[tri] (0, 0) -- (62.5, 1) -- (125, 0) -- cycle;
  \draw[tri] (62.5, 0) -- (125, 1) -- (187.5, 0) -- cycle;
  \draw[tri] (125, 0) -- (187.5, 1) -- (250, 0) -- cycle;
  \draw[tri] (187.5, 0) -- (250, 1) -- (343.75, 0) -- cycle;
  \draw[tri] (250, 0) -- (343.75, 1) -- (437.5, 0) -- cycle;
  \draw[tri] (343.75, 0) -- (437.5, 1) -- (562.5, 0) -- cycle;

  % Mid freq filters
  \draw[tri] (1000, 0) -- (1156, 1) -- (1343, 0) -- cycle;
  \draw[tri] (1718, 0) -- (1937, 1) -- (2187, 0) -- cycle;
  \draw[tri] (2812, 0) -- (3125, 1) -- (3468, 0) -- cycle;

  % High freq filters (wide, sparse)
  \draw[hightri] (4593, 0) -- (5031, 1) -- (5500, 0) -- cycle;
  \draw[hightri] (5500, 0) -- (6031, 1) -- (6593, 0) -- cycle;
  \draw[hightri] (6593, 0) -- (7250, 1) -- (8000, 0) -- cycle;

  % Annotations
  \node[ann, align=left, text=teal!90!black, anchor=west] at (350, 1.7) {
    \textbf{Low-Frequency Bands:}\\
    Narrow filters ($\Delta f \approx 62.5\text{ Hz}$)\\
    Spans $1\text{--}2$ FFT bins
  };

  \node[ann, align=right, text=purple!90!black, anchor=east] at (7800, 1.7) {
    \textbf{High-Frequency Bands:}\\
    Broad filters ($\Delta f > 1{,}000\text{ Hz}$)\\
    Integrates $40+$ FFT bins
  };
\end{scope}

% ==================== (b) Microscopic Comparison ====================
\begin{scope}[yshift=-3.2cm, x=0.038cm, y=1.0cm]
  \node[font=\bfseries\small, anchor=west] at (0, 1.9) 
    {(b) Discrete Bin Integration: Narrow Base ($m{=}0$) vs. Wide Pool ($m{=}79$)};

  % Low-Freq Zoom Box
  \draw[draw=gray!50, fill=gray!3, rounded corners=3pt] (-5, -1.0) rectangle (160, 1.5);
  \node[font=\scriptsize\bfseries, text=teal!90!black, anchor=north west] at (0, 1.4) 
    {Band $m{=}0$ ($0\text{--}125\text{ Hz}$): 3 FFT Bins};

  \foreach \i in {0, 1, 2, 3, 4} {
    \pgfmathsetmacro{\x}{\i * 31.25}
    \draw[fill=blue!10, draw=blue!60] (\x, -0.6) rectangle ++(28, 0.5);
    \node[font=\tiny] at (\x + 14, -0.35) {bin \i};
    \node[font=\tiny, text=gray!70, below] at (\x + 14, -0.6) {\pgfmathparse{\i*31.25}\pgfmathprintnumber{\pgfmathresult}};
  }
  \draw[tri, ultra thick] (0, 0) -- (62.5, 0.9) -- (125, 0) -- cycle;
  \node[font=\tiny\bfseries, text=teal!90!black] at (62.5, 1.05) {$W[0, k]$};

  % High-Freq Zoom Box
  \draw[draw=gray!50, fill=gray!3, rounded corners=3pt] (185, -1.0) rectangle (365, 1.5);
  \node[font=\scriptsize\bfseries, text=purple!90!black, anchor=north west] at (190, 1.4) 
    {Band $m{=}79$ ($6{,}593\text{--}8{,}000\text{ Hz}$): 45 FFT Bins};

  \draw[hightri, ultra thick] (195, 0) -- (280, 0.9) -- (355, 0) -- cycle;
  \node[font=\tiny\bfseries, text=purple!90!black] at (280, 1.05) {$W[79, k]$};

  \node[font=\tiny, align=center, text=black!75] at (275, 0.35) {
    Weighted sum pooling $45$ linear spectral bins\\
    $\sum_{k=211}^{256} W[79, k] \cdot P[k]$
  };
  \node[font=\tiny, text=gray!70, below] at (195, -0.6) {$6{,}593\text{ Hz}$};
  \node[font=\tiny, text=gray!70, below] at (280, -0.6) {$7{,}250\text{ Hz}$};
  \node[font=\tiny, text=gray!70, below] at (355, -0.6) {$8{,}000\text{ Hz}$};
\end{scope}

\end{tikzpicture}
