% Figure 1.6a: The Sliding Window Contract and Group Condemnation Rule
% Illustrates K = 3 words, K + 1 = 4 windows, 50% overlap (S = W/2), and word-window mapping.
\begin{tikzpicture}[
  >=Stealth,
  font=\small,
  wordbox/.style={draw=black!75, fill=blue!8, font=\footnotesize\bfseries, align=center, inner sep=3pt, rounded corners=1.5pt},
  winbox/.style={draw=teal!70!black, fill=teal!6, font=\scriptsize, align=center, inner sep=2.5pt, rounded corners=1.5pt},
  failwinbox/.style={draw=red!80!black, fill=red!10, font=\scriptsize\bfseries, align=center, inner sep=2.5pt, rounded corners=1.5pt},
  dimline/.style={<->, >=Stealth, semithick, black!70},
  guideline/.style={densely dashed, black!30, thin},
  ann/.style={font=\scriptsize, align=left}
]

  % Total width: 13.0 cm representing sentence length N
  \def\L{13.0}
  % For K = 3 words, K + 1 = 4 windows with 50% overlap:
  % N = W + 3*(W/2) = 2.5 W => W = N / 2.5 = 0.40 N = 5.2 cm
  % S = W / 2 = 0.20 N = 2.6 cm
  \def\W{5.2}
  \def\S{2.6}

  % Coordinate definitions for timeline
  \draw[->, thick, black!70] (0, 0) -- (\L + 0.6, 0) node[right, font=\scriptsize\bfseries] {Sentence Timeline $t \in [0, N]$};

  % Ticks at 0, 0.2N, 0.4N, 0.6N, 0.8N, 1.0N
  \foreach \x/\label in {0/{$0$}, 2.6/{$S = 0.2N$}, 5.2/{$2S = 0.4N$}, 7.8/{$3S = 0.6N$}, 10.4/{$4S = 0.8N$}, 13.0/{$N$}} {
    \draw[black!60] (\x, -0.1) -- (\x, 0.1);
    \node[below, font=\tiny, text=black!75] at (\x, -0.15) {\label};
    \draw[guideline] (\x, 0.1) -- (\x, 5.8);
  }

  % Row 1: The Spoken Words (K = 3 words)
  \node[anchor=west, font=\footnotesize\bfseries, text=blue!80!black] at (0, 5.5) {Spoken Words ($K = 3$ words across sentence):};

  \draw[wordbox] (0, 4.6) rectangle (4.29, 5.2) node[midway] {Word 1: ``READ''\\{\tiny ($0 \le t < 0.33N$)}};
  \draw[wordbox] (4.29, 4.6) rectangle (8.71, 5.2) node[midway] {Word 2: ``THE''\\{\tiny ($0.33N \le t < 0.67N$)}};
  \draw[wordbox] (8.71, 4.6) rectangle (13.0, 5.2) node[midway] {Word 3: ``BOOK''\\{\tiny ($0.67N \le t \le N$)}};

  % Acoustic boundary markers between words
  \draw[densely dashed, blue!70!black, thick] (4.29, 4.5) -- (4.29, 5.3) node[above, font=\tiny\bfseries, text=blue!80!black] {Boundary 1};
  \draw[densely dashed, blue!70!black, thick] (8.71, 4.5) -- (8.71, 5.3) node[above, font=\tiny\bfseries, text=blue!80!black] {Boundary 2};

  % Row 2: Overlapping Sliding Windows (K + 1 = 4 windows, S = W/2)
  \node[anchor=west, font=\footnotesize\bfseries, text=teal!80!black] at (0, 4.1) {Sliding Windows ($K+1 = 4$ windows, $W = \frac{2N}{K+2} = 0.40N$, Shift $S = \frac{W}{2} = 0.20N$):};

  % Window 1: y = 3.2
  \draw[winbox] (0, 3.2) rectangle (5.2, 3.75) 
    node[midway] {\textbf{Window 1} ($[0, W]$): Overlaps Word 1 and part of Word 2 $\quad [r_1 \ge 0.3 \implies \text{Pass}]$};

  % Window 2: y = 2.4 (Assume r < 0.3 to demonstrate Group Condemnation Rule)
  \draw[failwinbox] (2.6, 2.4) rectangle (7.8, 2.95) 
    node[midway] {\textbf{Window 2} ($[S, S+W]$): Overlaps Word 1 and Word 2 $\quad [\mathbf{r_2 < 0.3} \implies \textbf{FAIL}]$};

  % Window 3: y = 1.6
  \draw[winbox] (5.2, 1.6) rectangle (10.4, 2.15) 
    node[midway] {\textbf{Window 3} ($[2S, 2S+W]$): Overlaps Word 2 and Word 3 $\quad [r_3 \ge 0.3 \implies \text{Pass}]$};

  % Window 4: y = 0.8
  \draw[winbox] (7.8, 0.8) rectangle (13.0, 1.35) 
    node[midway] {\textbf{Window 4} ($[3S, 3S+W]$): Overlaps part of Word 2 and Word 3 $\quad [r_4 \ge 0.3 \implies \text{Pass}]$};

  % Sizing Dimension Lines
  \draw[dimline] (0, 0.35) -- (5.2, 0.35) node[midway, fill=white, font=\tiny\bfseries, inner sep=1.5pt] {$W = 0.40N$};
  \draw[dimline] (0, 0.15) -- (2.6, 0.15) node[midway, fill=white, font=\tiny\bfseries, inner sep=1.5pt] {$S = W/2 = 0.20N$};

  % Overlap callout
  \draw[<->, >=Stealth, red!70!black, semithick] (2.6, 3.05) -- (5.2, 3.05) node[midway, fill=white, font=\tiny\bfseries, text=red!80!black, inner sep=1pt] {$50\%$ Overlap};

\end{tikzpicture}
