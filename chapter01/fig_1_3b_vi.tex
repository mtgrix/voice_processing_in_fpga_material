% Figure 1.3b: 257 bin FFT tuyến tính sụp đổ dưới 80 bộ lọc tam giác Mel (Bản tiếng Việt)
\begin{tikzpicture}[
  >=Stealth,
  font=\small,
  tri/.style={draw=teal!85!black, thick, fill=teal!12},
  hightri/.style={draw=purple!85!black, thick, fill=purple!12},
  bin/.style={draw=black!35, thin},
  ax/.style={->, black!75, thick},
  tk/.style={font=\scriptsize, text=black!75},
  ann/.style={font=\footnotesize, text=black!85}
]

% ==================== (a) Toàn bộ dải phổ 0 - 8 kHz ====================
\begin{scope}[x=0.00205cm, y=1.2cm]
  \node[font=\bfseries\small, anchor=west] at (-120, 2.35) 
    {(a) Phân bổ ngân hàng bộ lọc Mel trên toàn dải âm thanh ($0\text{ đến } 8\text{ kHz}$)};

  % Trục tần số
  \draw[ax] (-80, 0) -- (8300, 0) node[right, font=\scriptsize\bfseries] {Tần số (Hz)};
  
  \foreach \f in {0, 1000, 2000, 3000, 4000, 5000, 6000, 7000, 8000} {
    \draw[black!60, semithick] (\f, -0.06) -- (\f, 0.06);
  }
  \node[tk, below] at (0, -0.06) {$0$};
  \node[tk, below] at (1000, -0.06) {$1\text{k}$};
  \node[tk, below] at (2000, -0.06) {$2\text{k}$};
  \node[tk, below] at (3000, -0.06) {$3\text{k}$};
  \node[tk, below] at (4000, -0.06) {$4\text{k}$};
  \node[tk, below] at (5000, -0.06) {$5\text{k}$};
  \node[tk, below] at (6000, -0.06) {$6\text{k}$};
  \node[tk, below] at (7000, -0.06) {$7\text{k}$};
  \node[tk, below] at (8000, -0.06) {$8\text{k}$};

  % Các bin FFT tuyến tính bên dưới (tổng cộng 257 bin)
  \fill[black!5] (0, 0) rectangle (8000, 0.10);
  \foreach \b in {0, 32, 64, 96, 128, 160, 192, 224, 256} {
    \pgfmathsetmacro{\hz}{\b * 31.25}
    \draw[bin] (\hz, 0) -- (\hz, 0.75);
    \node[font=\tiny, text=gray!70, anchor=south] at (\hz, 0.75) {$k{=}\b$};
  }

  % Các bộ lọc tam giác tiêu biểu
  % Dải tần thấp (dày đặc, hẹp)
  \draw[tri] (0, 0) -- (62.5, 0.45) -- (125, 0) -- cycle;
  \draw[tri] (62.5, 0) -- (125, 0.45) -- (187.5, 0) -- cycle;
  \draw[tri] (125, 0) -- (187.5, 0.45) -- (250, 0) -- cycle;
  \draw[tri] (187.5, 0) -- (250, 0.45) -- (343.75, 0) -- cycle;
  \draw[tri] (250, 0) -- (343.75, 0.45) -- (437.5, 0) -- cycle;
  \draw[tri] (343.75, 0) -- (437.5, 0.45) -- (562.5, 0) -- cycle;

  % Dải tần trung
  \draw[tri] (1000, 0) -- (1156, 0.45) -- (1343, 0) -- cycle;
  \draw[tri] (1718, 0) -- (1937, 0.45) -- (2187, 0) -- cycle;
  \draw[tri] (2812, 0) -- (3125, 0.45) -- (3468, 0) -- cycle;

  % Dải tần cao (rộng, thưa)
  \draw[hightri] (4593, 0) -- (5031, 0.45) -- (5500, 0) -- cycle;
  \draw[hightri] (5500, 0) -- (6031, 0.45) -- (6593, 0) -- cycle;
  \draw[hightri] (6593, 0) -- (7250, 0.45) -- (8000, 0) -- cycle;

  % Chú thích
  \node[ann, align=left, text=teal!90!black, anchor=west] at (350, 1.7) {
    \textbf{Các dải tần thấp:}\\
    Bộ lọc hẹp ($\Delta f \approx 62{,}5\text{ Hz}$)\\
    Chỉ phủ vài bin FFT kế cận
  };

  \node[ann, align=right, text=purple!90!black, anchor=east] at (7800, 1.7) {
    \textbf{Các dải tần cao:}\\
    Bộ lọc rộng ($\Delta f > 1.000\text{ Hz}$)\\
    Tích hợp hàng chục bin FFT
  };
\end{scope}

% ==================== (b) So sánh vi mô ====================
\begin{scope}[yshift=-3.1cm, y=1.0cm]
  \node[font=\bfseries\small, anchor=west] at (-0.3, 2.1) 
    {(b) Tích phân bin rời rạc: Đáy hẹp ($m{=}0$) so với Bể gom rộng ($m{=}79$)};

  % Hộp zoom tần thấp (bên trái, rộng 7.8cm)
  \draw[draw=gray!50, fill=gray!4, rounded corners=3pt] (-0.3, -0.9) rectangle (7.5, 1.8);
  \node[font=\footnotesize\bfseries, text=teal!90!black, anchor=north west] at (0, 1.7) 
    {Dải $m{=}0$ ($0\text{--}125\text{ Hz}$): Vài bin FFT};

  % Các bin FFT rời rạc trong hộp tần thấp
  \foreach \i/\freq in {0/0, 1/31.25, 2/62.5, 3/93.75, 4/125} {
    \pgfmathsetmacro{\bx}{0.2 + \i * 1.4}
    \draw[fill=blue!10, draw=blue!60] (\bx, -0.55) rectangle ++(1.2, 0.45);
    \node[font=\tiny] at (\bx + 0.6, -0.32) {bin \i};
    \node[font=\tiny, text=gray!70, below] at (\bx + 0.6, -0.55) {\freq};
  }
  % Tam giác cho Dải 0
  \draw[tri, ultra thick] (0.2, 0) -- (3.0, 0.6) -- (5.8, 0) -- cycle;
  \node[font=\scriptsize\bfseries, text=teal!90!black] at (3.0, 0.88) {$W[0, k]$};

  % Hộp zoom tần cao (bên phải, rộng 8.5cm)
  \draw[draw=gray!50, fill=gray!4, rounded corners=3pt] (8.2, -0.9) rectangle (16.7, 1.8);
  \node[font=\footnotesize\bfseries, text=purple!90!black, anchor=north west] at (8.5, 1.7) 
    {Dải $m{=}79$ ($6.593\text{--}8.000\text{ Hz}$): Hàng chục bin FFT};

  % Dải các bin trong hộp tần cao
  \foreach \j in {0, 1, 2, ..., 22} {
    \pgfmathsetmacro{\bx}{8.5 + \j * 0.35}
    \draw[fill=purple!8, draw=purple!35] (\bx, -0.55) rectangle ++(0.3, 0.45);
  }
  \node[font=\tiny, text=purple!80!black] at (12.45, -0.32) {$\dots$ dải dày đặc các bin phổ tuyến tính $\dots$};

  % Tam giác cho Dải 79
  \draw[hightri, ultra thick] (8.5, 0) -- (12.45, 0.65) -- (16.4, 0) -- cycle;
  \node[font=\scriptsize\bfseries, text=purple!90!black] at (12.45, 0.95) {$W[79, k]$};

  \node[font=\scriptsize, align=center, text=black!85] at (12.45, 0.25) {
    $\sum_{k} W[79, k] \cdot P[k]$
  };
  \node[font=\tiny, text=gray!70, below] at (8.5, -0.55) {$6.593\text{ Hz}$};
  \node[font=\tiny, text=gray!70, below] at (12.45, -0.55) {$7.250\text{ Hz}$};
  \node[font=\tiny, text=gray!70, below] at (16.4, -0.55) {$8.000\text{ Hz}$};
\end{scope}

\end{tikzpicture}
