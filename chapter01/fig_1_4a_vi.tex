% Figure 1.4a: Sàn trễ âm học vật lý (Bản tiếng Việt vẽ lại)
\begin{tikzpicture}[
  >=Stealth,
  font=\small,
  timeline/.style={->, thick, black!75},
  tk/.style={font=\footnotesize, text=black!85}
]

% Trục thời gian (0 đến 52 ms)
% Tỷ lệ: 0.29cm mỗi ms -> 50 ms = 14.5 cm
\begin{scope}[x=0.29cm, y=1.0cm]
  % Trục thời gian nằm ngang
  \draw[timeline] (0, 0) -- (53.5, 0) node[right, font=\footnotesize\bfseries] {Thời gian (ms)};

  % Vạch chia và nhãn số nằm ngang (không xoay)
  \foreach \t in {0, 10, 20, 25, 30, 40, 50} {
    \draw[black!60, semithick] (\t, -0.1) -- (\t, 0.1);
    \node[tk, below] at (\t, -0.12) {\t};
    \draw[gray!25, dashed] (\t, 0.1) -- (\t, 4.2);
  }

  % Đường gióng sàn âm học vật lý tại 25 ms
  \draw[red!80!black, very thick, dashed] (25, -0.2) -- (25, 4.2);

  % Callout một dòng duy nhất: "Khung 0 không tồn tại trước 25 ms vì mẫu 400 chưa tới."
  \draw[<-, very thick, red!85!black] (25.1, 3.65) -- (28.0, 3.65);
  \node[draw=red!75!black, fill=red!6, rounded corners=2pt, thick, inner sep=3.5pt, anchor=west] 
    at (28.0, 3.65) {
    \footnotesize\bfseries\textcolor{red!90!black}{Khung 0 không tồn tại trước $25\text{ ms}$ vì mẫu 400 chưa tới.}
  };

  % Thanh 1: Khung 0 (0 đến 25 ms, L = 400 mẫu)
  \draw[fill=blue!12, draw=blue!80!black, thick, rounded corners=2pt] (0, 2.5) rectangle (25, 3.25);
  \node[font=\footnotesize\bfseries, text=blue!90!black] at (12.5, 2.87) {
    Khung 0: Tích lũy vật lý $25\text{ ms}$ ($L = 400\text{ mẫu}$)
  };

  % Thanh 2: Khung 1 (10 đến 35 ms)
  % Đoạn gối đầu (10 đến 25 ms)
  \draw[fill=teal!18, draw=teal!80!black, thick, rounded corners=2pt] (10, 1.4) rectangle (25, 2.15);
  \node[font=\scriptsize, text=teal!95!black] at (17.5, 1.77) {
    Gối đầu $60\%$ ($240\text{ mẫu}$)
  };
  % Đoạn bước mới (25 đến 35 ms)
  \draw[fill=teal!32, draw=teal!80!black, thick, rounded corners=2pt] (25, 1.4) rectangle (35, 2.15);
  \node[font=\scriptsize\bfseries, text=teal!95!black] at (30.0, 1.77) {
    Bước mới ($160\text{ mẫu}$)
  };
  \node[font=\footnotesize\bfseries, text=teal!90!black, anchor=west] at (35.5, 1.77) {
    Khung 1 ($25\text{ ms}$)
  };

  % Thanh 3: Khung 2 (20 đến 45 ms)
  \draw[fill=purple!12, draw=purple!80!black, thick, rounded corners=2pt] (20, 0.3) rectangle (45, 1.05);
  \node[font=\footnotesize, text=purple!90!black] at (32.5, 0.67) {
    Khung 2: Nhịp bước tiếp theo ($L = 400\text{ mẫu}$)
  };
  \node[font=\footnotesize\bfseries, text=purple!90!black, anchor=west] at (45.5, 0.67) {
    Khung 2
  };

  % Móc nhọn nhịp bước 10 ms (25 đến 35 ms)
  \draw[decorate, decoration={brace, mirror, amplitude=4pt}, thick, black!75] (25, -0.55) -- (35, -0.55);
  \node[font=\footnotesize\bfseries, text=black!85, below=4pt] at (30, -0.55) {Nhịp bước $H/f_s = 10\text{ ms}$};

\end{scope}

\end{tikzpicture}
