% Sơ đồ kiến trúc vi mô phần cứng của module ch07_mac_engine.v

\begin{tikzpicture}[
    >=latex,
    font=\small,
    box/.style={draw=black!80, rounded corners=3pt, line width=0.8pt, fill=white, align=center, inner sep=5pt},
    fsmbox/.style={draw=green!60!black, rounded corners=3pt, line width=0.9pt, fill=green!5, align=center, inner sep=5pt},
    rambox/.style={draw=blue!70!black, rounded corners=3pt, line width=0.9pt, fill=blue!5, align=center, inner sep=5pt},
    dspbox/.style={draw=orange!80!black, rounded corners=3pt, line width=0.9pt, fill=orange!5, align=center, inner sep=5pt},
    regbox/.style={draw=red!70!black, rounded corners=2pt, line width=0.8pt, fill=red!10, align=center, inner sep=3pt},
    lbl/.style={font=\scriptsize, text=black!85, fill=white, inner sep=1.5pt},
    titlelbl/.style={font=\bfseries\small, text=black!90}
]

    % =========================================================================
    % 1. Khối FSM và Bộ phát địa chỉ (Bên trái)
    % =========================================================================
    \node[fsmbox, minimum width=3.4cm, minimum height=4.2cm] (fsm) at (1.7, 2.0) {
        \textbf{Bộ phát địa chỉ FSM}\\[4pt]
        \scriptsize Quét $98.540\text{ cặp}$\\
        \scriptsize $\tau = 0 \to n \in [0, 637]$\\
        \scriptsize $\tau = 26..214 \to n \in [0, 637-\tau]$\\[6pt]
        \scriptsize Bộ đếm $n$: \texttt{n\_cnt} (10 bit)\\
        \scriptsize Chỉ số lag: \texttt{lag\_idx} (8 bit)\\[6pt]
        \scriptsize \textbf{Cơ chế Zero-Bubble}\\
        \scriptsize Không chèn nhịp trống\\
        \scriptsize giữa các lag
    };

    % =========================================================================
    % 2. Bộ nhớ True Dual-Port RAM 638x17 (Giữa trên)
    % =========================================================================
    \node[rambox, minimum width=3.4cm, minimum height=2.2cm] (ram) at (6.6, 3.2) {
        \textbf{Bộ đệm mẫu $x_w$}\\[2pt]
        \scriptsize \texttt{ch07\_xw\_ram.v}\\
        \scriptsize 638 từ $\times$ 17-bit có dấu\\
        \scriptsize (2 khối M10K cấu hình 512$\times$20)\\[3pt]
        \scriptsize \textbf{Độ trễ đọc: 1 chu kỳ}
    };

    % =========================================================================
    % 3. Tầng đồng bộ cờ điều khiển (Giữa dưới)
    % =========================================================================
    \node[regbox, minimum width=3.2cm, minimum height=1.3cm] (pipe1) at (6.6, 0.7) {
        \textbf{Cân bằng nhịp RAM (1 nhịp)}\\[2pt]
        \scriptsize \texttt{ram\_vld\_r, ram\_sload\_r}\\
        \scriptsize \texttt{ram\_last\_r, ram\_tau\_r}
    };

    % =========================================================================
    % 4. Khối MAC DSP cứng (Bên phải)
    % =========================================================================
    \node[dspbox, minimum width=3.8cm, minimum height=3.8cm] (mac) at (12.0, 2.0) {
        \textbf{Khối MAC DSP}\\[2pt]
        \scriptsize \texttt{ch07\_mac\_unit.v}\\
        \scriptsize (ALTERA\_MULT\_ADD trong 1 DSP)\\[4pt]
        \scriptsize \textbf{Tầng 1:} Thanh ghi vào \texttt{a\_r, b\_r}\\
        \scriptsize \textbf{Nhân:} Bộ nhân cứng $17\times 17$ bit\\
        \scriptsize \textbf{Tầng 2:} Bộ cộng tích lũy 48-bit\\
        \scriptsize Nạp đè khi \texttt{sload\_r} = 1\\[4pt]
        \scriptsize \textbf{Độ trễ DSP: 2 chu kỳ}
    };

    % =========================================================================
    % Dây nối và tín hiệu
    % =========================================================================
    % Từ FSM sang RAM
    \draw[->, line width=0.9pt] (fsm.east |- ram.west) ++(0, 0.4) -- ++(1.5, 0)
        node[midway, above, lbl] {\texttt{addr\_a} ($n$)};
    \draw[->, line width=0.9pt] (fsm.east |- ram.west) ++(0, -0.4) -- ++(1.5, 0)
        node[midway, below, lbl] {\texttt{addr\_b} ($n+\tau$)};

    % Từ FSM sang Pipe1
    \draw[->, line width=0.8pt] (fsm.east |- pipe1.west) -- (pipe1.west)
        node[midway, above, lbl] {\texttt{sload, last, tau}};

    % Từ RAM sang MAC
    \draw[->, line width=0.9pt, blue!70!black] (ram.east) ++(0, 0.3) -- ++(2.0, 0)
        node[midway, above, lbl] {$x_w[n]$ (17 bit)};
    \draw[->, line width=0.9pt, blue!70!black] (ram.east) ++(0, -0.3) -- ++(2.0, 0)
        node[midway, below, lbl] {$x_w[n+\tau]$ (17 bit)};

    % Từ Pipe1 sang MAC
    \draw[->, line width=0.8pt] (pipe1.east) -- ++(0.7, 0) |- (mac.west) ++(0, -1.0)
        node[midway, below, lbl] {\texttt{sload\_r, last\_r}};

    % Tín hiệu ngõ ra từ MAC
    \draw[->, line width=1.1pt, orange!80!black] (mac.east) ++(0, 0.8) -- ++(1.4, 0)
        node[midway, above, lbl] {\texttt{out\_valid}};
    \draw[->, line width=1.1pt, orange!80!black] (mac.east) ++(0, 0.0) -- ++(1.4, 0)
        node[midway, above, lbl] {\texttt{out\_tau}};
    \draw[->, line width=1.1pt, orange!80!black] (mac.east) ++(0, -0.8) -- ++(1.4, 0)
        node[midway, above, lbl] {\texttt{out\_R} (48 bit)};

    % Chú thích nhịp thời gian dưới chân
    \node[draw=black!40, fill=black!2, rounded corners=3pt, inner sep=4pt, anchor=north] 
        at (6.8, -0.4) {
        \scriptsize\textbf{Tổng độ trễ đường ống:} 1 nhịp (RAM M10K) + 1 nhịp (Thanh ghi vào DSP) + 1 nhịp (Thanh ghi tổng DSP) = \textbf{3 chu kỳ}.\\
        \scriptsize Khung 638 mẫu: $98.540\text{ phát} + 3\text{ trễ} = \mathbf{98.543\text{ chu kỳ}}$ ($\approx 1{,}97\text{ ms}$ tại 50 MHz, $\text{RTF} = 0{,}197$).
    };

\end{tikzpicture}
