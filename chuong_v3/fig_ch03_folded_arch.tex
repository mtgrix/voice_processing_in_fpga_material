% Architecture of the 31-tap folded symmetric FIR filter with 17-bit pre-adder and 1 DSP block
\begin{tikzpicture}[>=stealth, font=\small]

    % Style definitions
    \tikzstyle{block}     = [draw=blue!80!black, fill=blue!6, thick, rounded corners=3pt, 
                             inner sep=5pt, align=center, font=\scriptsize\bfseries]
    \tikzstyle{regbox}    = [draw=teal!80!black, fill=teal!8, thick, rounded corners=3pt, 
                             inner sep=5pt, align=center, font=\scriptsize\bfseries]
    \tikzstyle{dspbox}    = [draw=red!80!black, fill=red!6, thick, rounded corners=4pt, 
                             inner sep=6pt, align=center, font=\scriptsize\bfseries]
    \tikzstyle{rombox}    = [draw=purple!80!black, fill=purple!6, thick, rounded corners=3pt, 
                             inner sep=5pt, align=center, font=\scriptsize\bfseries]
    \tikzstyle{muxbox}    = [draw=orange!80!black, fill=orange!12, thick, 
                             inner sep=4pt, align=center, font=\scriptsize\bfseries]
    \tikzstyle{fsmbox}    = [draw=gray!80!black, fill=gray!8, thick, rounded corners=4pt, 
                             inner sep=5pt, align=center, font=\scriptsize\bfseries]
    \tikzstyle{dataline}  = [->, very thick, color=teal!80!black]
    \tikzstyle{ctrlline}  = [->, thick, dashed, color=purple!80!black]
    \tikzstyle{portlabel} = [font=\scriptsize\ttfamily]

    % Outer Boundary Frame
    \draw[draw=blue!60!black, fill=blue!1, thick, rounded corners=6pt, dashed] 
        (-7.0, -4.8) rectangle (9.0, 4.3);
    \node[anchor=north west, font=\footnotesize\bfseries, text=blue!90!black, fill=blue!1, inner sep=2pt] 
        at (-6.7, 4.15) {Khối lọc FIR 31 tap (\texttt{ch03\_fir\_filter.v})};

    % Upstream Avalon-ST Sink Ports (Left)
    \node[portlabel, anchor=east] (sink_data)  at (-7.2,  1.5) {\texttt{sink\_data[15:0]}};
    \node[portlabel, anchor=east] (sink_valid) at (-7.2, -1.8) {\texttt{sink\_valid}};
    \node[portlabel, anchor=east] (sink_ready) at (-7.2, -2.4) {\texttt{sink\_ready}};
    \node[portlabel, anchor=east] (bypass_in)  at (-7.2, -3.2) {\texttt{bypass}};

    % Downstream Avalon-ST Source Ports (Right)
    \node[portlabel, anchor=west] (src_data)   at (9.2,  1.5) {\texttt{source\_data[15:0]}};
    \node[portlabel, anchor=west] (src_valid)  at (9.2, -1.8) {\texttt{source\_valid}};
    \node[portlabel, anchor=west] (src_ready)  at (9.2, -2.4) {\texttt{source\_ready}};

    % 1. Shift Register Block (31 stages x 16 bit)
    \node[regbox, minimum width=2.2cm, minimum height=1.4cm] (shift_reg) at (-4.9, 1.5)
        {Mảng thanh ghi dịch\\31 tầng $\times$ 16-bit\\(\texttt{shift\_reg[0..30]})};

    % 2. Pre-adder (17-bit signed)
    \node[block, minimum width=2.0cm, minimum height=1.4cm] (preadder) at (-1.0, 1.5)
        {Bộ tiền cộng\\Pre-adder 17-bit\\$x[k] + x[30-k]$};

    % 3. Coefficient ROM (16 x 16 bit)
    \node[rombox, minimum width=2.0cm, minimum height=1.2cm] (rom) at (-1.0, -0.7)
        {ROM hệ số Q1.15\\16 từ $\times$ 16-bit\\(\texttt{h[0..15]})};

    % 4. Multiplier (17 x 16 -> 33 bit inside DSP)
    \node[dspbox, minimum width=2.0cm, minimum height=1.4cm] (dsp_mult) at (2.6, 1.5)
        {Bộ nhân DSP\\17-bit $\times$ 16-bit\\Tích $p$ (33-bit)};

    % 5. Accumulator (40-bit)
    \node[regbox, minimum width=2.0cm, minimum height=1.2cm] (acc) at (2.6, -0.7)
        {Bộ tích lũy\\Thanh ghi 40-bit\\(\texttt{acc})};

    % 6. Rounding & Saturation
    \node[block, minimum width=1.8cm, minimum height=1.4cm] (round_sat) at (6.0, 1.5)
        {Làm tròn \&\\Bão hòa 16b\\$\gg 15$, ghim};

    % 7. Bypass Multiplexer
    \node[muxbox, minimum width=0.8cm, minimum height=1.8cm] (bypass_mux) at (7.8, 1.5)
        {MUX\\2:1};

    % 8. FSM Controller & Step Counter
    \node[fsmbox, minimum width=8.6cm, minimum height=1.2cm] (fsm) at (0.6, -3.2)
        {Bộ điều khiển FSM \& Bộ đếm bước 4-bit (\texttt{step[3:0]})\\Trạng thái: \texttt{S\_IDLE}, \texttt{S\_CALC}, \texttt{S\_WAIT\_OUT} (16 chu kỳ/mẫu)};

    % -------------------------------------------------------------
    % Data Connections
    % -------------------------------------------------------------
    % Input data to Shift Register
    \draw[dataline] (-7.0, 1.5) -- (shift_reg.west);

    % Bypass direct line (taps at x = -6.5, routes cleanly at y = 3.6)
    \fill (-6.5, 1.5) circle (2.2pt);
    \draw[dataline, color=orange!85!black] 
        (-6.5, 1.5) -- (-6.5, 3.6) -- (7.8, 3.6) -- (bypass_mux.north);
    \node[above=2pt, font=\scriptsize\bfseries, text=orange!90!black, fill=white, inner sep=2pt] 
        at (3.5, 3.6) {Đường Bypass trực tiếp (\texttt{bypass = 1})};
    \node[anchor=south, font=\scriptsize\bfseries, text=orange!90!black, fill=white, inner sep=1pt] 
        at (7.8, 2.58) {1};

    % Shift Register to Pre-adder (gap is 1.9 cm, label perfectly centered)
    \draw[dataline] (shift_reg.east) -- (preadder.west);
    \node[fill=white, text=teal!90!black, font=\scriptsize, inner sep=1.5pt] at (-2.95, 1.5) {Cặp mẫu};

    % Pre-adder to DSP Multiplier (input A)
    \draw[dataline] (preadder.east) -- ([yshift=0.25cm]dsp_mult.west);
    \node[fill=white, text=teal!90!black, font=\scriptsize, inner sep=1.5pt] at (0.8, 1.75) {17 bit};

    % ROM to DSP Multiplier (input B, wire at x = 0.5, label on the left)
    \draw[dataline] (rom.east) -- ++(0.5, 0) -- ++(0, 1.95) -- ([yshift=-0.25cm]dsp_mult.west);
    \node[left=2pt, font=\scriptsize, text=purple!90!black, fill=white, inner sep=1pt] at (0.5, 0.4) {$h[step]$ (16b)};

    % DSP Multiplier to Accumulator (p input, at x = 2.1)
    \draw[dataline] ([xshift=-0.5cm]dsp_mult.south) -- ([xshift=-0.5cm]acc.north);
    \node[left=2pt, font=\scriptsize, text=teal!90!black, fill=white, inner sep=1pt] at (2.1, 0.4) {Tích $p$ (33b)};

    % Accumulator feedback to Multiplier/Adder (at x = 3.1)
    \draw[dataline] ([xshift=0.5cm]acc.north) -- ([xshift=0.5cm]dsp_mult.south);
    \node[right=2pt, font=\scriptsize, text=teal!90!black, fill=white, inner sep=1pt] at (3.1, 0.4) {Cộng dồn};

    % Accumulator to Rounding & Saturation (at x = 4.6)
    \draw[dataline] (acc.east) -- (4.6, -0.7) -- (4.6, 1.5) -- (round_sat.west);
    \node[right=2pt, font=\scriptsize, text=teal!90!black, fill=white, inner sep=1pt] at (4.6, 0.4) {acc (40b)};

    % Rounding & Saturation to Bypass MUX (input 0)
    \draw[dataline] (round_sat.east) -- (bypass_mux.west);
    \node[anchor=east, font=\scriptsize\bfseries, fill=white, inner sep=1pt] at (7.35, 1.5) {0};
    \node[above=3pt, font=\scriptsize, text=teal!90!black, fill=white, inner sep=1.5pt] at (7.1, 1.5) {Lọc};

    % MUX output to source_data port
    \draw[dataline] (bypass_mux.east) -- (9.0, 1.5);

    % -------------------------------------------------------------
    % Control Connections (Dashed purple) - Clean parallel tracks
    % -------------------------------------------------------------
    % sink_valid input to FSM (track y = -1.8)
    \draw[ctrlline] (-7.0, -1.8) -- (-2.8, -1.8) -- (-2.8, -2.6);

    % sink_ready output from FSM (track y = -2.4)
    \draw[ctrlline] (-2.0, -2.6) -- (-2.0, -2.4) -- (-7.0, -2.4);

    % bypass input to FSM and MUX (track y = -3.2 and y = -4.3)
    \draw[ctrlline] (-7.0, -3.2) -- (fsm.west);
    \fill (-5.4, -3.2) circle (1.8pt);
    \draw[ctrlline, color=orange!80!black] 
        (-5.4, -3.2) -- (-5.4, -4.3) -- (7.8, -4.3) -- (bypass_mux.south);
    \node[fill=white, text=orange!90!black, font=\scriptsize, inner sep=2pt] 
        at (1.2, -4.3) {\texttt{bypass} (chọn kênh MUX 2:1)};

    % source_ready input to FSM (track y = -2.4)
    \draw[ctrlline] (9.0, -2.4) -- (3.8, -2.4) -- (3.8, -2.6);

    % source_valid output from FSM (track y = -1.8)
    \draw[ctrlline] (4.4, -2.6) -- (4.4, -1.8) -- (9.0, -1.8);

    % step[3:0] counter to ROM (straight vertical)
    \draw[ctrlline] (-1.0, -2.6) -- (rom.south);
    \node[fill=white, text=purple!90!black, font=\scriptsize, inner sep=1.5pt] at (-1.0, -1.95) {\texttt{step[3:0]}};

    % Latch output control from FSM to Round_Sat
    \draw[ctrlline] (2.6, -2.6) -- (2.6, -1.9) -- (6.0, -1.9) -- (round_sat.south);
    \node[fill=white, text=purple!90!black, font=\scriptsize, inner sep=1.5pt] at (4.3, -1.9) {\texttt{latch\_out}};

\end{tikzpicture}
