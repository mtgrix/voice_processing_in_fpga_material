    % =========================================================================

    \node[above=2pt, font=\scriptsize, text=red!80!black, align=center] at (opt1.north)

    \node[font=\scriptsize, align=center, text width=4.1cm] at (2.3, -0.5) {

        Tài nguyên: $\approx 350\text{ ALM}$ (không chốt)\\[3pt]
        \textbf{Không khả thi ở 50 MHz!}

        Thông lượng: \textbf{1 phép chia / chu kỳ}\\[3pt]
        $F_{\max} > 150\text{ MHz}$ (đạt 50 MHz)\\[3pt]
        Tài nguyên: $\approx 500\text{ ALM}$ (lãng phí!)\\[3pt]
        \emph{Thừa thông lượng (1 chia/khung)}
    };

    % =========================================================================
    % PHƯƠNG ÁN 3: BỘ CHIA TUẦN TỰ RADIX-2 (x = 13.2)

        $F_{\max} > 150\text{ MHz}$ (dư dả ở 50 MHz)\\[3pt]
        Tài nguyên: \textbf{$\approx 80\text{ ALM}$}, 0 DSP\\[3pt]
