% Microarchitecture of the 1st-order IIR DC-blocker filter (ch05_dc_blocker.v)
\begin{tikzpicture}[>=stealth, font=\small, scale=0.82, every node/.style={transform shape}]

    % Style definitions with constrained text width to guarantee ample arrow spacing
    \tikzstyle{block}     = [draw=blue!80!black, fill=blue!6, thick, rounded corners=4pt, 
                             inner sep=4pt, align=center, font=\small\bfseries]
    \tikzstyle{regbox}    = [draw=teal!80!black, fill=teal!8, thick, rounded corners=4pt, 
                             inner sep=4pt, align=center, font=\small\bfseries]
    \tikzstyle{calcbox}   = [draw=orange!80!black, fill=orange!8, thick, rounded corners=4pt, 
                             inner sep=4pt, align=center, font=\small\bfseries]
    \tikzstyle{muxbox}    = [draw=purple!80!black, fill=purple!8, thick, rounded corners=3pt, 
                             inner sep=4pt, align=center, font=\small\bfseries]
    \tikzstyle{logicbox}  = [draw=red!80!black, fill=red!6, thick, rounded corners=3pt, 
                             inner sep=4pt, align=center, font=\small\bfseries]
    \tikzstyle{dataline}  = [->, very thick, color=teal!80!black]
    \tikzstyle{ctrlline}  = [->, thick, dashed, color=red!80!black]
    \tikzstyle{portlabel} = [font=\footnotesize\ttfamily]

    % Frame boundary
    \draw[draw=blue!60!black, fill=blue!1, thick, rounded corners=6pt, dashed] 
        (-8.8, -4.6) rectangle (10.0, 5.3);
    \node[anchor=north west, font=\small\bfseries, text=blue!90!black, fill=white, draw=blue!60!black, rounded corners=2pt, inner sep=2.5pt] 
        at (-8.5, 5.1) {Kiến trúc bộ lọc triệt tiêu một chiều DC-Blocker (\texttt{ch05\_dc\_blocker.v})};

    % Left Ports (Sink Interface)
    \node[portlabel, anchor=east] (p_s_data)  at (-8.9,  2.2) {\texttt{sink\_data[15:0]}};
    \node[portlabel, anchor=east] (p_s_valid) at (-8.9,  0.8) {\texttt{sink\_valid}};
    \node[portlabel, anchor=east] (p_s_ready) at (-8.9, -0.6) {\texttt{sink\_ready}};
    \node[portlabel, anchor=east] (p_bypass)  at (-8.9, -3.9) {\texttt{bypass}};

    % Right Ports (Source Interface)
    \node[portlabel, anchor=west] (p_d_data)  at (10.1,  2.2) {\texttt{source\_data[15:0]}};
    \node[portlabel, anchor=west] (p_d_valid) at (10.1,  0.8) {\texttt{source\_valid}};
    \node[portlabel, anchor=west] (p_d_ready) at (10.1, -0.6) {\texttt{source\_ready}};

    % 1. Input Differentiator (Top Left)
    \node[regbox, text width=1.3cm, minimum height=1.0cm] (z_x) at (-5.6, 1.1)
        {Trễ $z^{-1}$\\$x[n-1]$};
    \node[calcbox, text width=1.3cm, minimum height=1.0cm] (sub_x) at (-3.8, 2.2)
        {Bộ trừ\\$dx[n]$};

    % 2. Scaling dx to Q20.16 (Mid Left)
    \node[calcbox, text width=1.5cm, minimum height=1.0cm] (shift_dx) at (-2.0, 2.2)
        {Dịch trái 16\\$dx \cdot 2^{16}$};

    % 3. Main Accumulator Adder (Center)
    \node[calcbox, text width=1.3cm, minimum height=1.0cm] (adder_w) at (-0.1, 2.2)
        {Bộ cộng\\$w[n]$};

    % 4. State Register z^-1 (Center Bottom)
    \node[regbox, text width=2.0cm, minimum height=1.1cm] (z_w) at (-0.1, -0.7)
        {Trạng thái $z^{-1}$\\36b Q20.16\\$w[n-1]$};

    % 5. Feedback Arithmetic Shift (Center Left Bottom)
    \node[calcbox, text width=2.0cm, minimum height=1.0cm] (shift_fb) at (-2.8, -0.7)
        {Nhân $\alpha = 1 - 2^{-8}$\\$w - (w \ggg 8)$};

    % 6. Rounding & Saturation (Right Center)
    \node[calcbox, text width=1.6cm, minimum height=1.0cm] (sat_box) at (3.6, 2.2)
        {Làm tròn\\kẹp int16};

    % 7. Bypass Multiplexer (Far Right)
    \node[muxbox, text width=1.2cm, minimum height=2.4cm] (mux) at (7.4, 2.2)
        {Mux\\$2:1$};

    % 8. Handshake logic OR gate (Bottom Center)
    \node[logicbox, text width=3.4cm, minimum height=1.0cm] (or_gate) at (0.4, -2.8)
        {Bắt tay Avalon-ST (OR gate)\\$\texttt{sink\_ready} = \texttt{source\_ready} \lor \overline{\texttt{valid}}$};

    % =========================================================================
    % SIGNAL INTERCONNECTIONS
    % =========================================================================

    % Sink data branch to z_x and sub_x
    \draw[dataline] (p_s_data) -- (-6.6, 2.2) |- (z_x.west);
    \draw[dataline] (-6.6, 2.2) -- (sub_x.west);
    \draw[dataline] (z_x.east) -| (sub_x.south);

    % sub_x to shift_dx to adder_w
    \draw[dataline] (sub_x.east) -- (shift_dx.west);
    \draw[dataline] (shift_dx.east) -- (adder_w.west);

    % adder_w to sat_box (2.4 cm generous arrow length, label has ample whitespace)
    \draw[dataline] (adder_w.east) -- (sat_box.west);
    \node[above=2pt, font=\footnotesize\bfseries, fill=white, inner sep=1.5pt] at (1.75, 2.2) {36b Q20.16};

    % State register feedback loop
    \draw[dataline] (adder_w.south) -- (z_w.north);
    \draw[dataline] (z_w.west) -- (shift_fb.east);
    \draw[dataline] (shift_fb.north) |- (adder_w.south west);

    % sat_box to Mux input 0 (2.2 cm empty arrow, label completely outside Mux)
    \draw[dataline] (sat_box.east) -- (mux.155);
    \node[above=2pt, font=\footnotesize\bfseries, text=purple!90!black, fill=white, inner sep=1.5pt] 
        at (5.4, 2.2) {$0$ (Lọc IIR)};

    % Bypass path from p_s_data to Mux input 1 (unclipped label placed cleanly above)
    \draw[dataline] (-6.6, 2.2) -- (-6.6, 4.0) -- (6.8, 4.0) |- (mux.25);
    \node[above=2pt, font=\footnotesize\bfseries, text=purple!90!black, fill=white, inner sep=1.5pt] 
        at (5.0, 4.0) {$1$ (Bypass ngõ vào)};

    % Bypass control: routes along very bottom at y = -3.9, clear of OR gate
    \draw[ctrlline] (p_bypass) -- (7.4, -3.9) -- (mux.south);

    % Mux to source_data
    \draw[dataline] (mux.east) -- (p_d_data);

    % =========================================================================
    % HANDSHAKE LOGIC: source_ready, sink_ready and Pipeline Clock Enable
    % =========================================================================

    % source_valid from mux/reg to source_valid port and OR gate
    \draw[ctrlline] (p_s_valid) -- (-7.2, 0.8) -- (-7.2, 0.0) -- (8.6, 0.0) -- (8.6, 0.8) -- (p_d_valid);
    \draw[ctrlline] (8.6, 0.0) |- (or_gate.north east); % Branch into OR gate as source_valid

    % source_ready enters from right port into OR gate
    \draw[ctrlline] (p_d_ready) -- (9.2, -0.6) -- (9.2, -2.8) -- (or_gate.east);

    % OR gate drives sink_ready to left port
    \draw[ctrlline] (or_gate.west) -- (-7.4, -2.8) -- (-7.4, -0.6) -- (p_s_ready);

    % OR gate output branches UP to gate pipeline registers (Clock Enable)
    \draw[ctrlline] (or_gate.north) -- (0.4, -1.6) -| (z_w.south);
    \draw[ctrlline] (0.4, -1.6) -| (z_x.south);
    \node[above=1pt, font=\footnotesize\bfseries, text=red!80!black, fill=white, inner sep=1.5pt] 
        at (-1.8, -1.6) {Cho phép chốt: \texttt{sink\_ready} (Clock Enable)};

\end{tikzpicture}
