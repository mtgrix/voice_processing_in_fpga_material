% Sơ đồ vi kiến trúc đường ống 3 tầng của khối quyết định vô thanh / hữu thanh (ch10_voicing)
% Phân tách rõ ràng giữa các tầng thanh ghi: Tầng 1 (Dịch-cộng), Tầng 2 (Nhân 54-bit), Tầng 3 (So sánh & MUX)
\begin{tikzpicture}[
    >=Stealth,
    scale=0.84,
    box_s1/.style={rectangle, draw=blue!80!black, fill=blue!5, rounded corners=2pt, minimum width=2.7cm, minimum height=0.88cm, align=center, font=\scriptsize},
    box_s2/.style={rectangle, draw=purple!80!black, fill=purple!5, rounded corners=2pt, minimum width=2.8cm, minimum height=1.10cm, align=center, font=\scriptsize},
    box_s3/.style={rectangle, draw=green!60!black, fill=green!5, rounded corners=2pt, minimum width=2.6cm, minimum height=0.88cm, align=center, font=\scriptsize},
    reg_bar/.style={rectangle, draw=black!80, fill=gray!35, minimum width=0.12cm, minimum height=5.3cm, inner sep=0pt},
    wire/.style={draw=black!80, thick, ->},
    bus_label/.style={font=\scriptsize, fill=white, inner sep=1.2pt}
]

    % -------------------------------------------------------------------------
    % TẦNG 1: DỊCH-CỘNG & TÍNH TOÁN TUYẾN TÍNH (Chu kỳ n)
    % -------------------------------------------------------------------------
    \node[box_s1] (b_term1) at (1.5, 4.9) {Dịch-cộng tính $t_1$:\\$100 S_c - 45 R_{0n}$};
    \node[box_s1] (b_mult3) at (1.5, 3.7) {Dịch-cộng hằng số:\\$3\cdot GP$ (18-bit)};
    \node[box_s1] (b_m100)  at (1.5, 2.5) {Dịch-cộng hằng số:\\$100\cdot R_{0n}$ (34-bit)};
    \node[box_s1] (b_ramp)  at (1.5, 1.3) {Trừ và bão hòa:\\$\max(0,\, 6GP - 145LP)$};
    \node[box_s1] (b_trap)  at (1.5, 0.1) {Phát hiện bẫy im lặng:\\$R_{0n}=0 \lor LP=0 \lor GP=0$};

    % Tín hiệu đầu vào dẫn tới Tầng 1
    \draw[wire] (-1.0, 5.1) -- node[above, bus_label] {$S_c\ [28]$} (b_term1.170);
    \draw[wire] (-1.0, 4.7) -- node[below, bus_label] {$R_{0n}\ [27]$} (b_term1.190);
    \draw[wire] (-1.0, 3.7) -- node[above, bus_label] {$GP\ [16]$} (b_mult3.west);
    \draw[wire] (-1.0, 2.5) -- node[above, bus_label] {$R_{0n}\ [27]$} (b_m100.west);
    \draw[wire] (-1.0, 1.3) -- node[above, bus_label] {$LP\ [16]$} (b_ramp.west);
    \draw[wire] (-1.0, 0.1) -- node[above, bus_label] {flags} (b_trap.west);

    % Thanh ghi Tầng 1 (reg 1)
    \node[reg_bar] (reg1) at (4.7, 2.65) {};
    \node[above=2pt, font=\scriptsize\bfseries, text=black!85] at (reg1.north) {reg 1};

    % Đường nối từ Khối Tầng 1 vào Thanh ghi 1
    \draw[wire] (b_term1.east) -- node[midway, above, bus_label] {$t_1\ [35]$} (4.64, 4.9);
    \draw[wire] (b_mult3.east) -- node[midway, above, bus_label] {$3GP\ [18]$} (4.64, 3.7);
    \draw[wire] (b_m100.east)  -- node[midway, above, bus_label] {$100R_{0n}\ [34]$} (4.64, 2.5);
    \draw[wire] (b_ramp.east)  -- node[midway, above, bus_label] {$ramp\ [18]$} (4.64, 1.3);
    \draw[wire] (b_trap.east)  -- node[midway, above, bus_label] {$trap$} (4.64, 0.1);

    % -------------------------------------------------------------------------
    % TẦNG 2: BỘ NHÂN TÍCH 54-BIT (Chu kỳ n+1)
    % -------------------------------------------------------------------------
    \node[box_s2] (b_mult_lhs) at (7.4, 4.3) {Bộ nhân tích $LHS$:\\$t_1 \times (3\cdot GP)$\\($35\times 18$ bit có dấu)};
    \node[box_s2] (b_mult_rhs) at (7.4, 1.7) {Bộ nhân tích $RHS$:\\$(100\cdot R_{0n}) \times ramp$\\($34\times 18$ bit không dấu)};

    % Nối từ Thanh ghi 1 vào Khối Tầng 2
    \draw[wire] (4.76, 4.6) -- (b_mult_lhs.165);
    \draw[wire] (4.76, 4.0) -- (b_mult_lhs.195);

    % Đường tín hiệu trap chạy qua Tầng 2 sang reg 2 (khoảng trống giữa hai khối nhân)
    \draw[wire] (4.76, 3.0) -- node[midway, above, bus_label] {$trap$} (10.04, 3.0);

    \draw[wire] (4.76, 2.0) -- (b_mult_rhs.165);
    \draw[wire] (4.76, 1.4) -- (b_mult_rhs.195);

    % Đường tín hiệu F0,raw chạy qua Tầng 2 sang reg 2 (dưới khối nhân RHS)
    \draw[wire] (4.76, 0.5) -- node[midway, above, bus_label] {$F_{0,\text{raw}}$} (10.04, 0.5);

    % Thanh ghi Tầng 2 (reg 2)
    \node[reg_bar] (reg2) at (10.1, 2.65) {};
    \node[above=2pt, font=\scriptsize\bfseries, text=black!85] at (reg2.north) {reg 2};

    % Nối từ Khối Tầng 2 vào Thanh ghi 2
    \draw[wire] (b_mult_lhs.east) -- node[midway, above, bus_label] {$LHS\ [54]$} (10.04, 4.3);
    \draw[wire] (b_mult_rhs.east) -- node[midway, above, bus_label] {$RHS\ [54]$} (10.04, 1.7);

    % -------------------------------------------------------------------------
    % TẦNG 3: SO SÁNH CHẶT & BỘ GHÉP KÊNH (Chu kỳ n+2)
    % -------------------------------------------------------------------------
    \node[box_s3] (b_comp) at (12.6, 4.3) {Bộ so sánh 54-bit:\\$LHS > RHS$};
    \node[box_s3] (b_gate) at (12.6, 2.7) {Cổng chặn im lặng:\\$voiced = (!trap) \ \& \ cmp$};
    \node[box_s3] (b_mux)  at (12.6, 1.1) {Bộ ghép kênh MUX:\\$voiced\ ?\ F_{0,\text{raw}} : 0$};

    % Nối từ Thanh ghi 2 vào Khối Tầng 3
    \draw[wire] (10.16, 4.5) -- node[midway, above, bus_label] {$LHS$} (b_comp.170);
    \draw[wire] (10.16, 4.1) -- node[midway, above, bus_label] {$RHS$} (b_comp.190);
    \draw[wire] (b_comp.south) -- node[midway, right=3pt, bus_label] {$cmp$} (b_gate.north);

    % Tín hiệu trap đi thẳng từ reg 2 vào b_gate
    \draw[wire] (10.16, 2.7) -- node[midway, above, bus_label] {$trap$} (b_gate.west);

    % Tín hiệu voiced từ b_gate đi xuống b_mux
    \draw[wire] (b_gate.south) -- node[midway, right=3pt, bus_label] {$voiced$} (b_mux.north);

    % Tín hiệu F0,raw đi từ reg 2 vào b_mux
    \draw[wire] (10.16, 1.1) -- node[midway, above, bus_label] {$F_{0,\text{raw}}$} (b_mux.west);

    % Thanh ghi ngõ ra Tầng 3 (reg 3)
    \node[reg_bar] (reg3) at (15.3, 2.65) {};
    \node[above=2pt, font=\scriptsize\bfseries, text=black!85] at (reg3.north) {reg 3};

    \draw[wire] (b_gate.east) -- node[midway, above, bus_label] {$out\_voiced$} (15.24, 2.7);
    \draw[wire] (b_mux.east)  -- node[midway, above, bus_label] {$F_{0,\text{gated}}$} (15.24, 1.1);
    \draw[wire] (10.16, 5.1)  -- node[midway, above, bus_label] {$valid\_pipe$} (15.24, 5.1);

    % Tín hiệu ra khỏi module
    \draw[wire] (15.36, 1.1) -- (16.7, 1.1) node[right, font=\scriptsize] {\texttt{out\_f0}};
    \draw[wire] (15.36, 2.7) -- (16.7, 2.7) node[right, font=\scriptsize] {\texttt{out\_voiced}};
    \draw[wire] (15.36, 5.1) -- (16.7, 5.1) node[right, font=\scriptsize] {\texttt{out\_valid}};

    % Nhãn chỉ dẫn chu kỳ trễ
    \node[above, font=\scriptsize\bfseries, text=blue!80!black]   at (1.5, 6.4) {Chu kỳ $n$: Tầng 1};
    \node[above, font=\scriptsize\bfseries, text=purple!80!black] at (7.4, 6.4) {Chu kỳ $n+1$: Tầng 2};
    \node[above, font=\scriptsize\bfseries, text=green!60!black]  at (12.6, 6.4) {Chu kỳ $n+2$: Tầng 3};
    \node[above, font=\scriptsize\bfseries, text=black!80]        at (17.2, 6.4) {Ngõ ra: $n+3$};
