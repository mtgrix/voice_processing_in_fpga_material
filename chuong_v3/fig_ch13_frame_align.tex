\begin{tikzpicture}[

    axis/.style={->, >=stealth, thick, line width=1.1pt},
    tick/.style={thick},

    annot/.style={font=\scriptsize, align=center, fill=white, draw=gray!30, rounded corners=2pt, inner sep=2.5pt}
]

% Time axis (scale: 2.0 units = 10 ms, so 1 ms = 0.2 units)
\draw[axis] (-0.5, 0) -- (14.5, 0) node[right, font=\small] {Thời gian $t$ (mili-giây)};

% Sample index axis (samples)
\draw[axis] (-0.5, -2.0) -- (14.5, -2.0) node[right, font=\small] {Chỉ số mẫu $n$ ($f_s = 16\text{ kHz}$)};

% Tick marks on time axis
\draw[tick] (0.0, 0.1) -- (0.0, -0.1) node[below=2pt, font=\scriptsize] {$0\text{ ms}$};
\draw[tick] (2.0, 0.1) -- (2.0, -0.1) node[below=2pt, font=\scriptsize] {$10\text{ ms}$};
\draw[tick] (4.0, 0.1) -- (4.0, -0.1) node[below=2pt, font=\scriptsize] {$t_1 \approx 20\text{ ms}^*$};
\draw[tick] (6.0, 0.1) -- (6.0, -0.1) node[below=2pt, font=\scriptsize] {$t_2 \approx 30\text{ ms}$};

% Vertical dashed center lines through frame centers down to time axis ticks
\draw[frameline, color=purple!70!black] (8.0, 4.4) -- (8.0, 0.1);
\draw[frameline, color=teal!70!black] (6.0, 3.3) -- (6.0, 0.1);
\draw[frameline, color=blue!70!black] (4.0, 2.2) -- (4.0, 0.1);

% Analysis windows drawn to exact scale: span = 39.875 ms (= 638/16000 s) = 7.975 units (~7.98 units)
% Frame 1: center t1 ~ 20 ms (x = 4.0), spans t1 +- 19.94 ms -> [0.1 ms, 39.97 ms] -> [0.02, 7.99]
\draw[winbar, fill=blue!15, draw=blue!70!black] (0.02, 1.3) rectangle (7.99, 2.1);
\node[font=\scriptsize, align=center, fill=blue!15, inner sep=1pt] at (4.0, 1.7) {
    \textbf{Khung 1} ($N = 638\text{ mẫu}$, $39{,}875\text{ ms} = 638/16000$)\\
    Tâm $t_1 \approx 20\text{ ms}$ (trải từ $0{,}1\text{ ms}$ đến $39{,}97\text{ ms}$)
};

% Frame 2: center t2 ~ 30 ms (x = 6.0), spans t2 +- 19.94 ms -> [10.1 ms, 49.97 ms] -> [2.02, 9.99]
\draw[winbar, fill=teal!15, draw=teal!70!black] (2.02, 2.4) rectangle (9.99, 3.2);
\node[font=\scriptsize, align=center, fill=teal!15, inner sep=1pt] at (6.0, 2.8) {
    \textbf{Khung 2} ($N = 638\text{ mẫu}$, $39{,}875\text{ ms} = 638/16000$)\\
    Tâm $t_2 \approx 30\text{ ms}$ (trải từ $10{,}1\text{ ms}$ đến $49{,}97\text{ ms}$)
};

% Frame 3: center t3 ~ 40 ms (x = 8.0), spans t3 +- 19.94 ms -> [20.1 ms, 59.97 ms] -> [4.02, 11.99]

\node[font=\scriptsize, align=center, fill=purple!15, inner sep=1pt] at (8.0, 3.9) {
    \textbf{Khung 3} ($N = 638\text{ mẫu}$, $39{,}875\text{ ms} = 638/16000$)\\
    Tâm $t_3 \approx 40\text{ ms}$ (trải từ $20{,}1\text{ ms}$ đến $59{,}97\text{ ms}$)
};

% Window span dimension arrow for Frame 1 (showing full 39.875 ms duration to scale)
\draw[<->, thick, draw=blue!80!black] (0.02, 0.75) -- (7.99, 0.75)
    node[midway, fill=white, inner sep=1.5pt, font=\scriptsize, text=blue!80!black] {Chiều dài cửa sổ phân tích $39{,}875\text{ ms}\ (=638/16000\text{ s})$};
\draw[dotted, thick, color=blue!60] (0.02, 1.3) -- (0.02, 0.65);

    node[midway, above=2pt, fill=white, inner sep=2pt, font=\scriptsize, text=red!70!black] {Bước nhảy thời gian $dt = 10\text{ ms}$ ($160\text{ mẫu}$)};

\draw[dotted, thick, color=gray!60] (0.02, -0.2) -- (0.02, -1.9);
\draw[dotted, thick, color=gray!60] (2.02, -0.2) -- (2.02, -1.9);

% Hop annotation between sample ticks

Mỗi cửa sổ phân tích $638\text{ mẫu}$ ($39{,}875\text{ ms} = 638/16000\text{ s}$) trải đối xứng hai bên tâm: $t_k \pm 19{,}94\text{ ms}$. Khung 1 trải từ $0{,}1\text{ ms}$ đến $39{,}97\text{ ms}$; Khung 2 từ $10{,}1\text{ ms}$ đến $49{,}97\text{ ms}$; Khung 3 từ $20{,}1\text{ ms}$ đến $59{,}97\text{ ms}$. Mẫu bắt đầu khung 1 là $s_1 = \text{first\_win0} = 1$ (ở $t \approx 0{,}06\text{ ms}$, không phải mẫu 0 do cơ chế định tâm đối xứng của Praat).
};
